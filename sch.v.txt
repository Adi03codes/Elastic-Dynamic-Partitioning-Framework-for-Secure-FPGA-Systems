`timescale 1ns / 1ps
//////////////////////////////////////////////////////////////////////////////////
// Project : Elastic Dynamic Partitioning Framework
// Module  : scheduler
//////////////////////////////////////////////////////////////////////////////////

module scheduler(

    input clk,
    input rst,

    input [3:0] request,

    output reg [1:0] grant,
    output reg valid

);

reg [1:0] pointer;

//---------------------------------------------------------
// Round Robin Scheduler
//---------------------------------------------------------

always @(posedge clk or posedge rst)
begin

    if(rst)
    begin
        pointer <= 2'd0;
        grant <= 2'd0;
        valid <= 1'b0;
    end

    else
    begin

        valid <= 1'b1;

        case(pointer)

        //-------------------------------------------------
        2'd0:
        begin
            if(request[0])
            begin
                grant <= 2'd0;
                pointer <= 2'd1;
            end
            else if(request[1])
            begin
                grant <= 2'd1;
                pointer <= 2'd2;
            end
            else if(request[2])
            begin
                grant <= 2'd2;
                pointer <= 2'd3;
            end
            else if(request[3])
            begin
                grant <= 2'd3;
                pointer <= 2'd0;
            end
            else
                valid <= 1'b0;
        end

        //-------------------------------------------------
        2'd1:
        begin
            if(request[1])
            begin
                grant <= 2'd1;
                pointer <= 2'd2;
            end
            else if(request[2])
            begin
                grant <= 2'd2;
                pointer <= 2'd3;
            end
            else if(request[3])
            begin
                grant <= 2'd3;
                pointer <= 2'd0;
            end
            else if(request[0])
            begin
                grant <= 2'd0;
                pointer <= 2'd1;
            end
            else
                valid <= 1'b0;
        end

        //-------------------------------------------------
        2'd2:
        begin
            if(request[2])
            begin
                grant <= 2'd2;
                pointer <= 2'd3;
            end
            else if(request[3])
            begin
                grant <= 2'd3;
                pointer <= 2'd0;
            end
            else if(request[0])
            begin
                grant <= 2'd0;
                pointer <= 2'd1;
            end
            else if(request[1])
            begin
                grant <= 2'd1;
                pointer <= 2'd2;
            end
            else
                valid <= 1'b0;
        end

        //-------------------------------------------------
        2'd3:
        begin
            if(request[3])
            begin
                grant <= 2'd3;
                pointer <= 2'd0;
            end
            else if(request[0])
            begin
                grant <= 2'd0;
                pointer <= 2'd1;
            end
            else if(request[1])
            begin
                grant <= 2'd1;
                pointer <= 2'd2;
            end
            else if(request[2])
            begin
                grant <= 2'd2;
                pointer <= 2'd3;
            end
            else
                valid <= 1'b0;
        end

        endcase

    end

end

endmodule
